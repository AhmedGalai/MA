\section*{CV Pipeline Steps}

\begin{longtable}{| >{\raggedright\arraybackslash}p{0.15\textwidth} | >{\raggedright\arraybackslash}p{0.5\textwidth} | >{\raggedright\arraybackslash}p{0.25\textwidth} |}
\hline
\textbf{Step} & \textbf{Description of how it works} & \textbf{Dependencies (e.g. files, libraries)} \\
\hline
\endhead

\textbf{1. Capture RealSense Depth} & Captures aligned RGB and depth frames from a RealSense camera. & \texttt{realsense\_depth.py}, \texttt{pyrealsense2} \\
\hline
\textbf{2. Update Headset Pose} & Receives and updates the headset's position and rotation, applying a probabilistic correction using a Kalman filter. & \texttt{pose\_manager.py}, \texttt{coordinate\_transformer.py} \\
\hline
\textbf{3. Transform Mask AVP -> RealSense} & Transforms the 2D mask from the Apple Vision Pro's view to the RealSense camera's view using a homography transformation. & \texttt{coordinate\_transformer.py} \\
\hline
\textbf{4. Estimate 6D Pose in RealSense View} & Estimates the 6D pose of the object using the depth map and the transformed mask. It can use a point cloud centroid/PCA approach or a PnP algorithm if a 3D model is available. & \texttt{pose\_estimator.py} \\
\hline
\textbf{5. Transform Pose RealSense -> AVP} & Transforms the estimated pose from the RealSense camera's coordinate system back to the Apple Vision Pro's coordinate system. & \texttt{coordinate\_transformer.py} \\
\hline
\textbf{(Optional) Calibration} & Performs a one-time calibration using ArUco markers to find the transformation between the headset and the RealSense camera. & \texttt{pose\_manager.py}, \texttt{cv2.aruco} \\
\hline
\end{longtable}

\section*{main\_api.py Endpoints}

\begin{longtable}{| >{\raggedright\arraybackslash}p{0.25\textwidth} | >{\raggedright\arraybackslash}p{0.1\textwidth} | >{\raggedright\arraybackslash}p{0.55\textwidth} |}
\hline
\textbf{Endpoint} & \textbf{Method} & \textbf{Description} \\
\hline
\endhead

\texttt{/health} & GET & Health check endpoint, returns status of the pipeline. \\
\hline
\texttt{/receive\_frame} & POST & Receives an RGB frame from an external source and stores it. \\
\hline
\texttt{/update\_head\_pose} & POST & Updates the headset's pose from the AVP. \\
\hline
\texttt{/process\_with\_mask} & POST & Processes a frame with a mask to estimate the 6D pose. \\
\hline
\texttt{/calibrate} & POST & Performs ArUco calibration. \\
\hline
\texttt{/stats} & GET & Get pipeline statistics. \\
\hline
\texttt{/avp\_frame} & GET & Get the latest AVP frame as a base64 encoded string. \\
\hline
\texttt{/mask} & GET & Get the latest mask as a base64 encoded string. \\
\hline
\texttt{/pose\_result} & GET & Get the latest pose estimation result. \\
\hline
\texttt{/head\_pose} & GET & Get the latest head pose. \\
\hline
\texttt{/intrinsics} & GET & Get the RealSense camera intrinsics. \\
\hline
\texttt{/pose\_history} & GET & Get the recent pose history. \\
\hline
\texttt{/shutdown} & POST & Shutdown the pipeline. \\
\hline
\end{longtable}
